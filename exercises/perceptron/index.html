    """Exercise 1 — Perceptron em dados separáveis.

Gera duas classes gaussianas 2D bem separadas, treina um perceptron escrito do
zero e mede a convergência. A classe ``Perceptron`` daqui é reutilizada, sem
alterações, pelo Exercise 2.

Uso (a partir da raiz do repositório):

    python docs/exercises/perceptron/code/exercise1_separable.py
"""

from pathlib import Path

import matplotlib.pyplot as plt
import numpy as np

FIGURES = Path(__file__).resolve().parents[1] / "figures"
RNG = np.random.default_rng(42)

N_PER_CLASS = 1000
ETA = 0.01
MAX_EPOCHS = 100

MEAN_0 = np.array([1.5, 1.5])
COV_0 = np.array([[0.5, 0.0], [0.0, 0.5]])
MEAN_1 = np.array([5.0, 5.0])
COV_1 = np.array([[0.5, 0.0], [0.0, 0.5]])


class Perceptron:
    """Perceptron de camada única com regra de atualização dirigida pelo erro."""

    def __init__(
        self,
        n_features: int,
        rng: np.random.Generator | None = None,
        eta: float = ETA,
        w_init: np.ndarray | None = None,
    ):
        # w_init permite repetir o treino com os MESMOS pesos iniciais e só outro eta
        # (item D do Exercise 1) — sem ele, cada instância sortearia pesos diferentes.
        self.w = np.array(w_init, dtype=float) if w_init is not None else rng.normal(0.0, 0.01, size=n_features)  # (1)!
        self.b = 0.0
        self.eta = eta

    def predict(self, X: np.ndarray) -> np.ndarray:
        return (X @ self.w + self.b >= 0).astype(int)

    def update(self, xi: np.ndarray, yi: int) -> bool:
        """Atualiza com uma amostra; devolve True se os pesos mudaram."""
        y_hat = int(xi @ self.w + self.b >= 0)
        error = yi - y_hat
        if error == 0:
            return False
        self.w = self.w + self.eta * error * xi
        self.b = self.b + self.eta * error
        return True

    def fit(self, X: np.ndarray, y: np.ndarray, max_epochs: int = MAX_EPOCHS) -> list[float]:
        """Treina até uma época sem atualização, ou max_epochs. Devolve acurácia/época."""
        accuracies = []
        self.updates_per_epoch = []  # guardado para a Análise (item D)
        for _ in range(max_epochs):
            updates = sum(self.update(xi, yi) for xi, yi in zip(X, y))
            self.updates_per_epoch.append(updates)
            accuracies.append(float((self.predict(X) == y).mean()))
            if updates == 0:
                break
        return accuracies


def generate_data() -> tuple[np.ndarray, np.ndarray]:
    class0 = RNG.multivariate_normal(MEAN_0, COV_0, size=N_PER_CLASS)
    class1 = RNG.multivariate_normal(MEAN_1, COV_1, size=N_PER_CLASS)
    X = np.vstack([class0, class1])
    y = np.concatenate([np.zeros(N_PER_CLASS, dtype=int), np.ones(N_PER_CLASS, dtype=int)])
    return X, y


def plot_figure1(X: np.ndarray, y: np.ndarray) -> None:
    fig, ax = plt.subplots(figsize=(6, 6))
    ax.scatter(*X[y == 0].T, s=10, alpha=0.6, label="Classe 0")
    ax.scatter(*X[y == 1].T, s=10, alpha=0.6, label="Classe 1")
    ax.set_xlabel("$x_1$")
    ax.set_ylabel("$x_2$")
    ax.set_title("Figura 1 — Dados separáveis")
    ax.legend()
    fig.tight_layout()
    fig.savefig(FIGURES / "fig01-separable-data.png", dpi=150)
    plt.close(fig)


def plot_decision_boundary(ax, model: Perceptron, X: np.ndarray) -> None:
    x1 = np.array([X[:, 0].min() - 0.5, X[:, 0].max() + 0.5])
    if abs(model.w[1]) > 1e-12:
        x2 = -(model.w[0] * x1 + model.b) / model.w[1]
        ax.plot(x1, x2, "k--", linewidth=1.5, label="Fronteira")
    else:
        x1_line = -model.b / model.w[0]
        ax.axvline(x1_line, color="k", linestyle="--", linewidth=1.5, label="Fronteira")


def plot_figure2(model: Perceptron, X: np.ndarray, y: np.ndarray) -> None:
    pred = model.predict(X)
    wrong = pred != y

    fig, ax = plt.subplots(figsize=(6, 6))
    ax.scatter(*X[y == 0].T, s=10, alpha=0.6, label="Classe 0")
    ax.scatter(*X[y == 1].T, s=10, alpha=0.6, label="Classe 1")
    ax.scatter(*X[wrong].T, s=60, facecolors="none", edgecolors="red", linewidths=1.5,
               label="Mal classificado")
    plot_decision_boundary(ax, model, X)
    ax.set_xlabel("$x_1$")
    ax.set_ylabel("$x_2$")
    ax.set_title("Figura 2 — Fronteira de decisão (dados separáveis)")
    ax.legend()
    fig.tight_layout()
    fig.savefig(FIGURES / "fig02-separable-boundary.png", dpi=150)
    plt.close(fig)


def plot_figure3(accuracies: list[float]) -> None:
    fig, ax = plt.subplots(figsize=(6, 4))
    ax.plot(range(1, len(accuracies) + 1), accuracies, marker="o", markersize=3)
    ax.set_xlabel("Época")
    ax.set_ylabel("Acurácia")
    ax.set_title("Figura 3 — Acurácia por época (dados separáveis)")
    fig.tight_layout()
    fig.savefig(FIGURES / "fig03-separable-accuracy.png", dpi=150)
    plt.close(fig)


def main() -> None:
    FIGURES.mkdir(parents=True, exist_ok=True)
    X, y = generate_data()
    plot_figure1(X, y)

    model = Perceptron(n_features=2, rng=RNG, eta=ETA)
    w_init = model.w.copy()  # guardado para repetir o treino com o mesmo início (item D)
    accuracies = model.fit(X, y)
    plot_figure2(model, X, y)
    plot_figure3(accuracies)

    print(f"eta={ETA}: w final = {model.w}, b final = {model.b:.4f}")
    print(f"épocas até convergir: {len(accuracies)}")
    print(f"acurácia final: {accuracies[-1]:.4f} ({accuracies[-1] * 100:.2f}%)")
    print(f"atualizações por época: {model.updates_per_epoch}")
    direction_001 = model.w / np.linalg.norm(model.w)
    print(f"direção de w (eta=0.01): {direction_001}")

    # Item D: repetir com eta = 1.0 e os MESMOS pesos iniciais — "sem mudar mais nada".
    model_eta1 = Perceptron(n_features=2, eta=1.0, w_init=w_init)
    accuracies_eta1 = model_eta1.fit(X, y)
    direction_eta1 = model_eta1.w / np.linalg.norm(model_eta1.w)
    print(f"\neta=1.0: w final = {model_eta1.w}, b final = {model_eta1.b:.4f}")
    print(f"épocas até convergir: {len(accuracies_eta1)}")
    print(f"acurácia final: {accuracies_eta1[-1]:.4f} ({accuracies_eta1[-1] * 100:.2f}%)")
    print(f"direção de w (eta=1.0): {direction_eta1}")
    print(f"produto escalar entre as duas direções: {direction_001 @ direction_eta1:.6f}")


if __name__ == "__main__":
    main()

    D — Análise

    Por que dados separáveis convergem rápido? Ligue sua resposta à regra de atualização: o que acontece com o número de atualizações por época ao longo do treino?

    Como as duas nuvens têm uma margem grande entre si em relação ao espalhamento, a maioria dos pontos já nasce do lado certo da fronteira inicial (quase aleatória, pesos ~0). A regra só atualiza em cima de erro (y - y_hat ≠ 0), então a cada época a fronteira se aproxima mais da região separadora e sobra cada vez menos pontos errados para corrigir. Isso aparece nos números: as atualizações por época caem de [3, 3, 4, 4, 3, 4, 3, 4, 2, 4, 2, 4, 2, 3, 3, 3, 2, 3, 3, 2, 3, 3, 2, 3, 1, 0] — uma tendência de queda (com ruído, porque cada atualização desloca a fronteira e pode "descorrigir" outro ponto) até chegar a 0 na época 26, quando o laço para.

    Re-execute o treino com eta = 1.0, sem mudar mais nada. Reporte a contagem de épocas e a acurácia final e compare a direção de w (ou seja, w / ||w||) com a da execução com eta = 0.01.

    Com eta = 1.0 (mesmos pesos iniciais que a execução com eta = 0.01 — só o eta muda): w = [5,871, 3,359], b = -31,00, convergiu em 37 épocas, acurácia final 100%.

    eta = 0.01 eta = 1.0
    w final [0,0505, 0,0289] [5,871, 3,359]
    b final -0,2500 -31,00
    Épocas 26 37
    Direção w/‖w‖ [0,8681, 0,4963] [0,8679, 0,4967]

    As duas direções são praticamente idênticas (produto escalar = 0,999999... ≈ 1) — ou seja, as duas execuções aprendem a mesma fronteira geometricamente, só que representada por vetores de magnitude diferente. Isso mostra o que eta controla: o tamanho do passo a cada atualização (eta * x, com ‖x‖ da ordem de poucas unidades aqui), não a direção para onde o perceptron converge. Um eta maior dá passos maiores — por isso são necessárias contagens de época diferentes (26 vs. 37, sem relação simples entre si, já que cada execução corrige um conjunto de erros ligeiramente diferente ao longo do caminho) — mas ambos escalam a mesma solução até ela classificar tudo corretamente.

    Agora argumente o que teria acontecido partindo de w = 0, b = 0. Mostre algebricamente que rodar todo o treino duas vezes, com eta_1 e eta_2, produz pesos que diferem apenas pelo fator constante eta_2 / eta_1.

    Partindo de w = 0, b = 0, a primeira predição é degrau(0) = 1 para toda amostra, qualquer que seja eta — o sinal da soma ponderada não depende de eta ainda. Logo o erro na primeira rodada (y - 1, que é 0 para a Classe 1 e -1 para a Classe 0) é o mesmo para qualquer execução, e a primeira atualização é w_1 = eta * (erro_1 * x_1), b_1 = eta * erro_1 — exatamente eta vezes o que seria a atualização de uma execução "base" com eta_base = 1 (chame os pesos dessa execução base de w~, b~).

    Por indução: suponha que, depois de k atualizações, w_k = eta * w~_k e b_k = eta * b~_k (verdadeiro para k=1). A predição da amostra seguinte depende só do sinal de w_k · x + b_k = eta * (w~_k · x + b~_k). Como eta > 0 não muda o sinal de um número, essa predição é idêntica à que a execução base teria — logo o erro na amostra k+1 também é idêntico, e a atualização k+1 é de novo eta vezes a atualização base. Por indução, em qualquer ponto do treino, w = eta * w~ e b = eta * b~.

    Isso vale para qualquer eta > 0: rodando o treino inteiro com eta_1 e de novo com eta_2, partindo ambos de zero, obtemos w_{eta_1} = eta_1 * w~ e w_{eta_2} = eta_2 * w~ — logo w_{eta_2} = (eta_2 / eta_1) * w_{eta_1}, diferem só por esse fator constante. Como o sinal (e portanto toda predição, toda contagem de erro e toda condição de parada) não muda com esse fator, a fronteira de decisão e o número de épocas ficam idênticos — eta não teria efeito algum. É exatamente por isso que o item B exige começar de pesos não nulos.

    """Exercise 2 — Perceptron em dados sobrepostos (algoritmo pocket).

Reutiliza a classe ``Perceptron`` do Exercise 1 sem alterações. Como os dados
não são separáveis, o laço nunca para de atualizar — este script acompanha,
além dos pesos finais, os melhores pesos vistos durante o treino (pocket).

Uso (a partir da raiz do repositório):

    python docs/exercises/perceptron/code/exercise2_pocket.py
"""

import sys
from pathlib import Path

import matplotlib.pyplot as plt
import numpy as np

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE))
from exercise1_separable import ETA, MAX_EPOCHS, Perceptron  # noqa: E402

FIGURES = HERE.parents[0] / "figures"
RNG = np.random.default_rng(42)

N_PER_CLASS = 1000
MEAN_0 = np.array([3.0, 3.0])
COV_0 = np.array([[1.5, 0.0], [0.0, 1.5]])
MEAN_1 = np.array([4.0, 4.0])
COV_1 = np.array([[1.5, 0.0], [0.0, 1.5]])


def generate_data() -> tuple[np.ndarray, np.ndarray]:
    class0 = RNG.multivariate_normal(MEAN_0, COV_0, size=N_PER_CLASS)
    class1 = RNG.multivariate_normal(MEAN_1, COV_1, size=N_PER_CLASS)
    X = np.vstack([class0, class1])
    y = np.concatenate([np.zeros(N_PER_CLASS, dtype=int), np.ones(N_PER_CLASS, dtype=int)])
    return X, y


def train_with_pocket(model: Perceptron, X: np.ndarray, y: np.ndarray, max_epochs: int = MAX_EPOCHS):
    """Mesmo laço de atualização do Exercise 1; só acrescenta a cópia do 'bolso'."""
    current_accuracies, pocket_accuracies = [], []
    best_acc = -1.0
    pocket_w, pocket_b = model.w.copy(), model.b
    pocket_epoch = 0

    for epoch in range(1, max_epochs + 1):
        for xi, yi in zip(X, y):
            changed = model.update(xi, yi)
            if changed:
                acc = float((model.predict(X) == y).mean())
                if acc > best_acc:
                    best_acc = acc
                    pocket_w, pocket_b = model.w.copy(), model.b
                    pocket_epoch = epoch
        current_accuracies.append(float((model.predict(X) == y).mean()))
        pocket_accuracies.append(best_acc)

    return pocket_w, pocket_b, pocket_epoch, current_accuracies, pocket_accuracies


def plot_figure4(X: np.ndarray, y: np.ndarray) -> None:
    fig, ax = plt.subplots(figsize=(6, 6))
    ax.scatter(*X[y == 0].T, s=10, alpha=0.6, label="Classe 0")
    ax.scatter(*X[y == 1].T, s=10, alpha=0.6, label="Classe 1")
    ax.set_xlabel("$x_1$")
    ax.set_ylabel("$x_2$")
    ax.set_title("Figura 4 — Dados sobrepostos")
    ax.legend()
    fig.tight_layout()
    fig.savefig(FIGURES / "fig04-overlapping-data.png", dpi=150)
    plt.close(fig)


def plot_figure5(model_final: Perceptron, model_pocket: Perceptron, X: np.ndarray, y: np.ndarray) -> None:
    wrong_pocket = model_pocket.predict(X) != y

    fig, ax = plt.subplots(figsize=(6, 6))
    ax.scatter(*X[y == 0].T, s=10, alpha=0.5, label="Classe 0")
    ax.scatter(*X[y == 1].T, s=10, alpha=0.5, label="Classe 1")
    ax.scatter(*X[wrong_pocket].T, s=40, facecolors="none", edgecolors="red", linewidths=1.2,
               label="Mal classificado (pocket)")

    x1 = np.array([X[:, 0].min() - 0.5, X[:, 0].max() + 0.5])
    x2_final = -(model_final.w[0] * x1 + model_final.b) / model_final.w[1]
    x2_pocket = -(model_pocket.w[0] * x1 + model_pocket.b) / model_pocket.w[1]
    ax.plot(x1, x2_final, "k:", linewidth=1.5, label="Fronteira final")
    ax.plot(x1, x2_pocket, "k--", linewidth=1.5, label="Fronteira pocket")

    ax.set_xlabel("$x_1$")
    ax.set_ylabel("$x_2$")
    ax.set_title("Figura 5 — Fronteiras final vs. pocket")
    ax.legend(fontsize=8)
    fig.tight_layout()
    fig.savefig(FIGURES / "fig05-overlapping-boundaries.png", dpi=150)
    plt.close(fig)


def plot_figure6(current_accuracies: list[float], pocket_accuracies: list[float]) -> None:
    fig, ax = plt.subplots(figsize=(6, 4))
    epochs = range(1, len(current_accuracies) + 1)
    ax.plot(epochs, current_accuracies, label="Pesos atuais")
    ax.plot(epochs, pocket_accuracies, label="Melhor até agora (pocket)")
    ax.set_xlabel("Época")
    ax.set_ylabel("Acurácia")
    ax.set_title("Figura 6 — Acurácia por época (dados sobrepostos)")
    ax.legend()
    fig.tight_layout()
    fig.savefig(FIGURES / "fig06-overlapping-accuracy.png", dpi=150)
    plt.close(fig)


def main() -> None:
    FIGURES.mkdir(parents=True, exist_ok=True)
    X, y = generate_data()
    plot_figure4(X, y)

    model = Perceptron(n_features=2, rng=RNG, eta=ETA)
    pocket_w, pocket_b, pocket_epoch, current_acc, pocket_acc = train_with_pocket(model, X, y)

    model_pocket = Perceptron(n_features=2, eta=ETA, w_init=pocket_w)
    model_pocket.b = pocket_b

    plot_figure5(model, model_pocket, X, y)
    plot_figure6(current_acc, pocket_acc)

    print(f"Pesos finais:  w = {model.w}, b = {model.b:.4f}, acurácia = {current_acc[-1]:.4f} "
          f"({current_acc[-1] * 100:.2f}%)")
    print(f"Pesos pocket:  w = {pocket_w}, b = {pocket_b:.4f}, acurácia = {pocket_acc[-1]:.4f} "
          f"({pocket_acc[-1] * 100:.2f}%)")
    print(f"Época em que o pocket foi atualizado pela última vez: {pocket_epoch}")


if __name__ == "__main__":
    main()

    D — Análise

    A melhor reta para estes dados acerta cerca de 73%. Seus pesos do pocket devem chegar perto disso; os finais, não. Explique a diferença. Onde a fronteira final fica em relação à nuvem de pontos e por que o laço de treino a deixa ali?

    O pocket chegou a 72,85%, bem perto do teto teórico de ~73% — é a melhor fronteira linear que o perceptron encontrou ao longo do treino. Os pesos finais, em vez disso, ficam com acurácia de 50,05%, essencialmente equivalente a chutar: como os dados nunca ficam 100% corretos, o laço segue atualizando para sempre, e a última atualização não tem motivo para ser boa — ela só reflete o erro da última amostra malclassificada visitada antes do limite de 100 épocas, não uma fronteira otimizada. Usando a dica do enunciado: com ‖x‖ ≈ 5 nestes dados, cada atualização de erro move b por ±eta = ±0,01, mas move w por ±eta·x, cuja norma é ≈ eta·5 = 0,05 — ou seja, w "anda" cerca de 5× mais que b por atualização. Depois de milhares de atualizações com direções de erro que se cancelam parcialmente (já que os dois lados da fronteira têm pontos errados), a direção de w fica instável e sem convergir a lugar nenhum estável, enquanto b fica comparativamente "preso" perto de zero — resultado: a fronteira final fica mal posicionada em relação à nuvem, cortando as duas classes de forma essencialmente arbitrária.

    Compare a Figura 3 com a Figura 6. No Exercício 1 a curva de acurácia estabiliza; aqui, não. O que o teorema da convergência do perceptron garante e qual das suas hipóteses este dataset viola?

    O teorema da convergência do perceptron garante que, se as classes forem linearmente separáveis, o algoritmo encontra uma fronteira que classifica tudo corretamente em um número finito de atualizações. A Figura 3 (Exercise 1) mostra exatamente essa garantia se cumprindo: a acurácia sobe e estabiliza em 100%. A Figura 6 (Exercise 2) nunca estabiliza porque este dataset viola a hipótese de separabilidade linear — as duas nuvens se sobrepõem, então não existe nenhuma reta com 0% de erro, e o teorema simplesmente não se aplica: não há garantia nenhuma de parada, e de fato ela não acontece (a curva "atual" oscila em torno de 50% do início ao fim).

    Mais épocas resolvem? Um eta menor resolve? Justifique pela regra de atualização e não por tentativa e erro.

    Não, nenhum dos dois resolve. A regra de atualização só para de mexer nos pesos quando uma época inteira não encontra nenhum erro — e como as classes se sobrepõem, sempre vai existir pelo menos um ponto do lado errado de qualquer reta, então sempre haverá pelo menos uma atualização por época, para sempre. Mais épocas só adiam o limite, não mudam essa condição. Um eta menor também não ajuda: como mostrado no Exercise 1 (item D), eta só reescala o tamanho do passo — ele não muda quais amostras estão erradas nem cria uma fronteira que não existia antes. O problema não é de ajuste fino de hiperparâmetro, é estrutural: o perceptron (um classificador linear) não tem capacidade de representar uma fronteira melhor que o teto de ~73% para dados que não são linearmente separáveis.

    Discussão

    O ponto mais contraintuitivo foi ver a acurácia dos pesos finais do Exercise 2 cair para perto de 50% em vez de estabilizar em algo próximo do pocket — reforça que, para dados não separáveis, "deixar treinar mais" não é uma estratégia válida sem um mecanismo extra (como o próprio pocket) para lembrar do melhor resultado já visto.

    Conclusão

    Os dois exercícios mostram o perceptron nos dois lados do teorema de convergência: com dados separáveis (Exercise 1), a regra de atualização converge rápido e de forma estável, e a taxa de aprendizado só afeta a escala dos pesos, não a fronteira aprendida. Com dados sobrepostos (Exercise 2), a mesma regra nunca para de atualizar, e os pesos "finais" perdem qualquer relação com uma boa fronteira — o algoritmo pocket existe precisamente para recuperar, de um treino que nunca converge, o melhor resultado que ele já visitou.

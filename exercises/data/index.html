    """Exercise 1 — Nuvens de pontos: geometria e espalhamento em 2D.

Gera as 4 classes gaussianas do enunciado em 4 escalas de espalhamento, salva
as Figuras 1-3 em ``figures/`` e imprime os números que alimentam a tabela
*Results summary* do relatório (razão de separação e taxa de mistura).

Uso (a partir da raiz do repositório):

    python docs/exercises/data/code/exercise1_point_clouds.py
"""

from itertools import combinations
from pathlib import Path

import matplotlib.pyplot as plt
import numpy as np

FIGURES = Path(__file__).resolve().parents[1] / "figures"
RNG = np.random.default_rng(42)  # (1)!

CLASSES = {
    0: {"mean": np.array([2.0, 3.0]), "std": np.array([0.8, 2.5])},
    1: {"mean": np.array([5.0, 6.0]), "std": np.array([1.2, 1.9])},
    2: {"mean": np.array([8.0, 1.0]), "std": np.array([0.9, 0.9])},
    3: {"mean": np.array([15.0, 4.0]), "std": np.array([0.5, 2.0])},
}
N_PER_CLASS = 100
SCALES = (0.5, 1.0, 2.0, 4.0)


def generate(scale: float) -> tuple[np.ndarray, np.ndarray]:
    """Amostra 100 pontos por classe; as médias não mudam, só o espalhamento."""
    xs, ys = [], []
    for label, params in CLASSES.items():
        xs.append(RNG.normal(params["mean"], params["std"] * scale, size=(N_PER_CLASS, 2)))
        ys.append(np.full(N_PER_CLASS, label))
    return np.vstack(xs), np.concatenate(ys)


def separation_ratios(scale: float) -> dict[tuple[int, int], float]:
    """r_ij = ||mu_i - mu_j|| / (sigma_bar_i + sigma_bar_j), por par de classes."""
    sigma_bar = {c: (p["std"] * scale).mean() for c, p in CLASSES.items()}
    ratios = {}
    for i, j in combinations(CLASSES, 2):
        dist = np.linalg.norm(CLASSES[i]["mean"] - CLASSES[j]["mean"])
        ratios[(i, j)] = dist / (sigma_bar[i] + sigma_bar[j])
    return ratios


def mixing_rate(X: np.ndarray, y: np.ndarray) -> float:
    """Fração de pontos cujo centro de classe mais próximo não é o verdadeiro."""
    means = np.stack([CLASSES[c]["mean"] for c in sorted(CLASSES)])
    dists = np.linalg.norm(X[:, None, :] - means[None, :, :], axis=2)
    nearest = dists.argmin(axis=1)
    return float((nearest != y).mean())


def bisector_segment(p1: np.ndarray, p2: np.ndarray, length: float = 7.0) -> np.ndarray:
    """Segmento da bissetriz perpendicular entre dois centros — esboço de fronteira."""
    midpoint = (p1 + p2) / 2
    direction = p2 - p1
    perp = np.array([-direction[1], direction[0]])
    perp = perp / np.linalg.norm(perp)
    return np.stack([midpoint - perp * length / 2, midpoint + perp * length / 2])


def plot_figure1(X: np.ndarray, y: np.ndarray) -> None:
    fig, ax = plt.subplots(figsize=(7, 5))
    for c in CLASSES:
        ax.scatter(*X[y == c].T, s=14, alpha=0.75, label=f"Classe {c}")
    centers = np.stack([CLASSES[c]["mean"] for c in CLASSES])
    ax.scatter(*centers.T, marker="x", s=120, c="black", linewidths=2, label="Centro")

    # Esboço (item C): fronteira aproximada entre pares de classes vizinhas,
    # a bissetriz perpendicular do segmento que une os dois centros.
    for i, j in ((0, 1), (1, 2), (2, 3)):
        seg = bisector_segment(CLASSES[i]["mean"], CLASSES[j]["mean"])
        ax.plot(*seg.T, "k--", linewidth=1.3, alpha=0.8)
    ax.plot([], [], "k--", linewidth=1.3, label="Fronteira aproximada (esboço)")

    ax.set_xlabel("$x_1$")
    ax.set_ylabel("$x_2$")
    ax.set_title("Figura 1 — Nuvens gaussianas (scale = 1.0)")
    ax.legend(loc="upper left", fontsize=8)
    fig.tight_layout()
    fig.savefig(FIGURES / "fig01-point-clouds.png", dpi=150)
    plt.close(fig)  # (2)!


def plot_figure2(datasets: dict[float, tuple[np.ndarray, np.ndarray]]) -> None:
    all_x = np.vstack([X for X, _ in datasets.values()])
    margin = 1.0
    xlim = (all_x[:, 0].min() - margin, all_x[:, 0].max() + margin)
    ylim = (all_x[:, 1].min() - margin, all_x[:, 1].max() + margin)

    fig, axes = plt.subplots(1, 4, figsize=(18, 4.5), sharex=True, sharey=True)
    for ax, scale in zip(axes, SCALES):
        X, y = datasets[scale]
        for c in CLASSES:
            ax.scatter(*X[y == c].T, s=10, alpha=0.7)
        ax.set_xlim(xlim)
        ax.set_ylim(ylim)
        ax.set_title(f"scale = {scale}")
        ax.set_xlabel("$x_1$")
    axes[0].set_ylabel("$x_2$")
    fig.suptitle("Figura 2 — Espalhamento das 4 classes por fator de escala")
    fig.tight_layout()
    fig.savefig(FIGURES / "fig02-scale-comparison.png", dpi=150)
    plt.close(fig)


def plot_figure3(rates: dict[float, float]) -> None:
    fig, ax = plt.subplots(figsize=(6, 4))
    ax.plot(list(rates.keys()), list(rates.values()), marker="o")
    ax.set_xlabel("Fator de escala $s$")
    ax.set_ylabel("Taxa de mistura")
    ax.set_title("Figura 3 — Taxa de mistura vs. fator de escala")
    fig.tight_layout()
    fig.savefig(FIGURES / "fig03-mixing-rate.png", dpi=150)
    plt.close(fig)


def main() -> None:
    FIGURES.mkdir(parents=True, exist_ok=True)

    datasets = {scale: generate(scale) for scale in SCALES}

    plot_figure1(*datasets[1.0])
    plot_figure2(datasets)

    rates = {scale: mixing_rate(*datasets[scale]) for scale in SCALES}
    plot_figure3(rates)

    ratios = separation_ratios(1.0)
    print("Razão de separação r_ij (s = 1.0):")
    for (i, j), r in sorted(ratios.items(), key=lambda item: item[1]):
        print(f"  par ({i}, {j}): r_ij = {r:.3f}")
    min_pair, min_r = min(ratios.items(), key=lambda item: item[1])
    print(f"Menor r_ij: par {min_pair} = {min_r:.3f}")
    print(f"Menor r_ij projetado em s=2.0 (r_ij(1)/2): {min_r / 2:.3f}")

    print("\nTaxa de mistura por escala:")
    for scale, rate in rates.items():
        print(f"  scale={scale:>4}: {rate:.4f} ({rate * 100:.2f}%)")


if __name__ == "__main__":
    main()

Isso é consistente com a taxa de mistura medida (Figura 3): ela já é 7,25% em s=1, mas salta para 19,25% em s=2 — o crescimento mais acentuado acontece justamente na faixa em que r_01 cruza 1.

    C — Análise

    Descreva a sobreposição das quatro classes no dataset original (s = 1). Uma única fronteira linear separaria todas as classes? E um conjunto de fronteiras lineares?

    Em s=1, as Classes 0 e 1 já se sobrepõem visivelmente (menor r_ij do conjunto, 1,326) por causa do desvio vertical grande da Classe 0 (sigma_y = 2,5); as Classes 1-2 e 2-3 têm alguma sobreposição nas bordas; a Classe 3 fica isolada à direita (r_03 = 4,496, o maior par). Uma única reta não separa as 4 classes — elas não estão dispostas ao longo de uma direção comum. Um conjunto de 3 retas (uma entre cada par de classes vizinhas, como na Figura 1) já separa a maior parte dos pontos corretamente, mas não todos: com s=1 a taxa de mistura já é 7,25% mesmo usando a fronteira ótima (mais próxima do centro), porque as gaussianas se sobrepõem em probabilidade, não apenas em posição.

    Esboce sobre a Figura 1 as fronteiras de decisão que você acha que uma rede neural treinada poderia aprender.

    Feito na Figura 1 (linhas tracejadas): cada segmento é a bissetriz perpendicular entre os centros de duas classes vizinhas — a fronteira que um classificador de "centro mais próximo" (a mesma regra usada para calcular a taxa de mistura) aprenderia.

    Relacione o esboço com o item B: quanto mais espalhadas as nuvens, o que acontece com a região onde a rede necessariamente erra?

    As fronteiras esboçadas não mudam de posição com s (dependem só das médias, que são fixas), mas a nuvem de pontos ao redor delas fica mais larga conforme s cresce. Ou seja, a região de erro inevitável — a faixa de cada lado da fronteira onde pontos de uma classe caem do lado da outra — cresce proporcionalmente ao espalhamento. Isso é exatamente o que a Figura 3 mostra: 0,25% → 7,25% → 19,25% → 48,25% conforme s vai de 0,5 a 4,0.

    """Exercise 2 — Não-linearidade em dimensões maiores.

Gera o Dataset I (gaussianas 5D deslocadas, correlacionadas) e o Dataset II
(cascas concêntricas 5D), projeta os dois em 2D via PCA e mede a distância
entre centros e o raio de cada ponto. Salva as Figuras 4-5 em ``figures/``.

Uso (a partir da raiz do repositório):

    python docs/exercises/data/code/exercise2_nonlinearity.py
"""

from pathlib import Path

import matplotlib.pyplot as plt
import numpy as np
from sklearn.decomposition import PCA

FIGURES = Path(__file__).resolve().parents[1] / "figures"
RNG = np.random.default_rng(42)

N_PER_CLASS = 500

# --- Dataset I: gaussianas multivariadas deslocadas ------------------------
MU_A = np.zeros(5)
COV_A = np.array(
    [
        [1.0, 0.8, 0.1, 0.0, 0.0],
        [0.8, 1.0, 0.3, 0.0, 0.0],
        [0.1, 0.3, 1.0, 0.5, 0.0],
        [0.0, 0.0, 0.5, 1.0, 0.2],
        [0.0, 0.0, 0.0, 0.2, 1.0],
    ]
)
MU_B = np.full(5, 1.5)
COV_B = np.array(
    [
        [1.5, -0.7, 0.2, 0.0, 0.0],
        [-0.7, 1.5, 0.4, 0.0, 0.0],
        [0.2, 0.4, 1.5, 0.6, 0.0],
        [0.0, 0.0, 0.6, 1.5, 0.3],
        [0.0, 0.0, 0.0, 0.3, 1.5],
    ]
)

# --- Dataset II: cascas concêntricas ----------------------------------------
# O enunciado dá o raio como rho ~ N(mu, 0.4); interpretamos 0.4 como o
# desvio padrão (não a variância) do raio de cada casca.
RADIUS_STD = 0.4
RADIUS_CORE = 2.0
RADIUS_SHELL = 5.0


def make_dataset_i() -> tuple[np.ndarray, np.ndarray]:
    a = RNG.multivariate_normal(MU_A, COV_A, size=N_PER_CLASS)
    b = RNG.multivariate_normal(MU_B, COV_B, size=N_PER_CLASS)
    X = np.vstack([a, b])
    y = np.concatenate([np.zeros(N_PER_CLASS), np.ones(N_PER_CLASS)])
    return X, y


def random_unit_vectors(n: int, dim: int = 5) -> np.ndarray:
    v = RNG.normal(size=(n, dim))
    return v / np.linalg.norm(v, axis=1, keepdims=True)


def make_dataset_ii() -> tuple[np.ndarray, np.ndarray]:
    u_core = random_unit_vectors(N_PER_CLASS)
    u_shell = random_unit_vectors(N_PER_CLASS)
    r_core = RNG.normal(RADIUS_CORE, RADIUS_STD, size=N_PER_CLASS)
    r_shell = RNG.normal(RADIUS_SHELL, RADIUS_STD, size=N_PER_CLASS)
    core = r_core[:, None] * u_core
    shell = r_shell[:, None] * u_shell
    X = np.vstack([core, shell])
    y = np.concatenate([np.zeros(N_PER_CLASS), np.ones(N_PER_CLASS)])
    return X, y


def center_distance(X: np.ndarray, y: np.ndarray) -> float:
    mean0 = X[y == 0].mean(axis=0)
    mean1 = X[y == 1].mean(axis=0)
    return float(np.linalg.norm(mean0 - mean1))


def plot_figure4(datasets: dict[str, tuple[np.ndarray, np.ndarray]]) -> dict[str, np.ndarray]:
    fig, axes = plt.subplots(1, 2, figsize=(11, 5))
    explained = {}
    for ax, (name, (X, y)) in zip(axes, datasets.items()):
        pca = PCA(n_components=2, random_state=42)
        proj = pca.fit_transform(X)
        for c, label in ((0, "Classe 0"), (1, "Classe 1")):
            ax.scatter(*proj[y == c].T, s=12, alpha=0.7, label=label)
        ax.set_title(name)
        ax.set_xlabel("PC1")
        ax.set_ylabel("PC2")
        ax.legend()
        explained[name] = pca.explained_variance_ratio_
    fig.suptitle("Figura 4 — Projeção PCA (2D) dos dois datasets 5D")
    fig.tight_layout()
    fig.savefig(FIGURES / "fig04-pca-projection.png", dpi=150)
    plt.close(fig)
    return explained


def plot_figure5(datasets: dict[str, tuple[np.ndarray, np.ndarray]]) -> None:
    fig, axes = plt.subplots(1, 2, figsize=(11, 4.5))
    for ax, (name, (X, y)) in zip(axes, datasets.items()):
        radius = np.linalg.norm(X, axis=1)
        for c, label in ((0, "Classe 0"), (1, "Classe 1")):
            ax.hist(radius[y == c], bins=25, alpha=0.6, label=label)
        ax.set_title(name)
        ax.set_xlabel(r"$\|x\|$")
        ax.set_ylabel("Contagem")
        ax.legend()
    fig.suptitle("Figura 5 — Histograma do raio por classe")
    fig.tight_layout()
    fig.savefig(FIGURES / "fig05-radius-histogram.png", dpi=150)
    plt.close(fig)


def main() -> None:
    FIGURES.mkdir(parents=True, exist_ok=True)

    datasets = {
        "Dataset I (gaussianas)": make_dataset_i(),
        "Dataset II (cascas)": make_dataset_ii(),
    }

    explained = plot_figure4(datasets)
    plot_figure5(datasets)

    for name, (X, y) in datasets.items():
        dist = center_distance(X, y)
        var2 = explained[name][:2].sum()
        print(f"{name}:")
        print(f"  distância entre centros (5D) = {dist:.4f}")
        print(f"  variância explicada PC1+PC2  = {var2:.4f} ({var2 * 100:.2f}%)")


if __name__ == "__main__":
    main()

    D — Análise

    No Dataset II, a distância entre os centros é próxima de zero, mas os histogramas de raio ficam bem separados. O que essa combinação diz sobre a possibilidade de separar as classes com um hiperplano?

    Diz que não é possível separar as classes do Dataset II com um hiperplano. Um hiperplano w^T x + b = 0 divide o espaço em dois lados a partir da posição média dos pontos ao longo da direção w — e como os centros das duas classes praticamente coincidem (distância 0,267, contra 3,228 no Dataset I), não existe direção w ao longo da qual as médias das duas classes se separem. A informação que distingue as classes está na distância à origem, uma quantidade que qualquer projeção linear apaga.

    Explique por que a estrutura do Dataset II não pode ser resolvida por uma fronteira linear, por mais dados que se colete.

    Por construção, o núcleo (raio ≈ 2,0) e a casca (raio ≈ 5,0) são duas cascas esféricas concêntricas em torno da origem. A fronteira real que as separa é a esfera ||x|| = c (para algum c entre 2 e 5) — uma condição quadrática nas coordenadas (sum_i x_i^2 = c^2), não linear. Coletar mais dados reduz o ruído de amostragem, mas não muda a geometria: qualquer hiperplano vai cortar as duas cascas ao meio, errando metade dos pontos de cada lado, não importa quantas amostras existam.

    A PCA é uma transformação linear. Discuta: uma projeção 2D em que as classes parecem misturadas prova que elas são inseparáveis no espaço original? Justifique com os seus próprios resultados e escreva uma função simples das entradas que separe o Dataset II.

    Não prova. A Figura 4 mostra as classes do Dataset II bastante misturadas na projeção PCA — mas isso é uma limitação da projeção, não do dado: a PCA escolhe as direções de maior variância, que aqui não coincidem com a direção discriminativa (o raio). A prova está na Figura 5: no espaço original (5D), o histograma de ||x|| separa as duas classes quase perfeitamente, algo que a PCA, sendo linear, não consegue enxergar. Uma função simples que separa o Dataset II é

    g(x) = sum_{i=1}^{5} x_i^2 = ||x||^2

    classificando como núcleo se g(x) < 14,5 (ponto médio entre 2^2 = 4 e 5^2 = 25) e como casca caso contrário.

    """Exercise 3 — Preparando dados reais (Spaceship Titanic) para uma rede tanh.

Requer o arquivo ``train.csv`` da competição Kaggle "Spaceship Titanic"
(https://www.kaggle.com/competitions/spaceship-titanic/data), colocado em
``docs/exercises/data/code/data/train.csv`` (baixe manualmente — é preciso
login no Kaggle; não há como baixar por script sem credenciais).

Uso (a partir da raiz do repositório):

    python docs/exercises/data/code/exercise3_preprocessing.py
"""

from pathlib import Path

import matplotlib.pyplot as plt
import numpy as np
import pandas as pd
from sklearn.impute import SimpleImputer
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import MinMaxScaler, OneHotEncoder

HERE = Path(__file__).resolve().parent
FIGURES = HERE.parents[0] / "figures"
DATA_PATH = HERE / "data" / "train.csv"

NUMERIC_COLS = ["Age", "RoomService", "FoodCourt", "ShoppingMall", "Spa", "VRDeck"]
SPEND_COLS = ["RoomService", "FoodCourt", "ShoppingMall", "Spa", "VRDeck"]
CATEGORICAL_COLS = ["HomePlanet", "CryoSleep", "Destination", "VIP"]
DROP_COLS = ["Cabin", "Name", "PassengerId"]
TARGET = "Transported"


def load_data() -> pd.DataFrame:
    if not DATA_PATH.exists():
        raise FileNotFoundError(
            f"Não encontrei {DATA_PATH}.\n"
            "Baixe train.csv em https://www.kaggle.com/competitions/spaceship-titanic/data "
            "e salve nesse caminho antes de rodar este script."
        )
    return pd.read_csv(DATA_PATH)


def describe_data(df: pd.DataFrame) -> None:
    positive_rate = df[TARGET].mean()
    print(f"Proporção da classe positiva em '{TARGET}': {positive_rate:.4f} ({positive_rate * 100:.2f}%)")

    missing = df.isna().sum()
    missing_pct = (missing / len(df) * 100).round(2)
    missing_table = pd.DataFrame({"faltantes": missing, "percentual": missing_pct})
    missing_table = missing_table[missing_table["faltantes"] > 0].sort_values("faltantes", ascending=False)
    print("\nValores faltantes por coluna:")
    print(missing_table)

    print("\nMédia, mediana e máximo das colunas de gasto (antes de transformar):")
    print(df[SPEND_COLS].agg(["mean", "median", "max"]).T)


def split_data(df: pd.DataFrame) -> tuple[pd.DataFrame, pd.DataFrame]:
    train_df, test_df = train_test_split(
        df, test_size=0.2, stratify=df[TARGET], random_state=42
    )
    return train_df, test_df


def preprocess(train_df: pd.DataFrame, test_df: pd.DataFrame) -> tuple[np.ndarray, np.ndarray, pd.DataFrame, pd.DataFrame]:
    train_df = train_df.drop(columns=DROP_COLS).copy()
    test_df = test_df.drop(columns=DROP_COLS).copy()

    # --- imputação: mediana para numéricas, moda para categóricas ----------
    num_imputer = SimpleImputer(strategy="median")
    train_df[NUMERIC_COLS] = num_imputer.fit_transform(train_df[NUMERIC_COLS])
    test_df[NUMERIC_COLS] = num_imputer.transform(test_df[NUMERIC_COLS])

    cat_imputer = SimpleImputer(strategy="most_frequent")
    train_df[CATEGORICAL_COLS] = cat_imputer.fit_transform(train_df[CATEGORICAL_COLS])
    test_df[CATEGORICAL_COLS] = cat_imputer.transform(test_df[CATEGORICAL_COLS])

    # --- feature de engenharia: gasto total ---------------------------------
    train_df["TotalSpend"] = train_df[SPEND_COLS].sum(axis=1)
    test_df["TotalSpend"] = test_df[SPEND_COLS].sum(axis=1)

    # --- cauda pesada: log1p nas colunas de gasto (+ TotalSpend) ------------
    heavy_tail_cols = SPEND_COLS + ["TotalSpend"]
    foodcourt_before = train_df["FoodCourt"].copy()
    for col in heavy_tail_cols:
        train_df[col] = np.log1p(train_df[col])
        test_df[col] = np.log1p(test_df[col])
    foodcourt_after = train_df["FoodCourt"].copy()

    # --- one-hot nas categóricas; handle_unknown ignora categoria nova ------
    encoder = OneHotEncoder(handle_unknown="ignore", sparse_output=False)
    train_cat = encoder.fit_transform(train_df[CATEGORICAL_COLS])
    test_cat = encoder.transform(test_df[CATEGORICAL_COLS])
    cat_names = encoder.get_feature_names_out(CATEGORICAL_COLS)

    # --- escalonamento para [-1, 1], compatível com a saída de tanh --------
    scale_cols = NUMERIC_COLS + ["TotalSpend"]
    scaler = MinMaxScaler(feature_range=(-1, 1))
    train_num = scaler.fit_transform(train_df[scale_cols])
    test_num = scaler.transform(test_df[scale_cols])

    X_train = np.hstack([train_num, train_cat])
    X_test = np.hstack([test_num, test_cat])
    feature_names = scale_cols + list(cat_names)

    plot_figure6(foodcourt_before, foodcourt_after)

    print(f"\nShape final da matriz de features de treino: {X_train.shape}")
    print(f"Shape final da matriz de features de teste:   {X_test.shape}")
    print(f"NaN remanescentes no treino: {np.isnan(X_train).sum()}")
    print(f"NaN remanescentes no teste:  {np.isnan(X_test).sum()}")
    print(f"Treino  — min={X_train.min():.4f}, max={X_train.max():.4f}")
    print(f"Teste   — min={X_test.min():.4f}, max={X_test.max():.4f}")
    print(f"Features: {feature_names}")

    return X_train, X_test, train_df, test_df


def plot_figure6(before: pd.Series, after: pd.Series) -> None:
    fig, axes = plt.subplots(1, 2, figsize=(11, 4.5))
    axes[0].hist(before, bins=40, color="tab:blue")
    axes[0].set_title("FoodCourt — antes (bruto)")
    axes[0].set_xlabel("FoodCourt")
    axes[0].set_ylabel("Contagem")

    axes[1].hist(after, bins=40, color="tab:orange")
    axes[1].set_title("FoodCourt — depois (log1p)")
    axes[1].set_xlabel("log(1 + FoodCourt)")
    axes[1].set_ylabel("Contagem")

    fig.suptitle("Figura 6 — Efeito do log1p em uma feature de cauda pesada")
    fig.tight_layout()
    fig.savefig(FIGURES / "fig06-foodcourt-log.png", dpi=150)
    plt.close(fig)


def main() -> None:
    FIGURES.mkdir(parents=True, exist_ok=True)
    df = load_data()
    describe_data(df)
    train_df, test_df = split_data(df)
    print(f"\nAmostras de treino: {len(train_df)} | Amostras de teste: {len(test_df)}")
    preprocess(train_df, test_df)


if __name__ == "__main__":
    main()

    Cauda pesada: aplique log(1+x) e mostre o histograma antes e depois. Por que essa transformação ajuda uma rede com tanh?

    Ver Figura 6. tanh satura (derivada perto de zero) para entradas com módulo grande — colunas de gasto brutas, com valores até ~30.000, empurrariam a maior parte dos neurônios para a região saturada, dificultando o aprendizado. O log1p comprime a cauda longa para uma escala muito menor e mais simétrica antes do escalonamento final.

    Escalonamento: implemente uma das duas opções, explique a escolha e reporte os valores mínimo e máximo resultantes.

    Escolhi MinMaxScaler para [-1, 1] (em vez de padronização com média 0 e desvio 1) porque o intervalo de saída bate exatamente com a imagem de tanh, o que a padronização não garante. Resultado, ajustado só no treino:

Isso é esperado (o scaler não pode conhecer o teste) e um lembrete de que "compatível com tanh" é aproximado, não uma garantia absoluta fora da amostra de treino.

A escolha de imputar com mediana/moda tem efeito menor porque as taxas de faltantes são baixas (~2% por coluna).

    Discussão

    O item mais delicado foi decidir o desvio-padrão do raio nas cascas concêntricas do Dataset II — o enunciado escreve ρ ~ N(2.0, 0.4), que pode ser lido como média/desvio-padrão ou média/variância; assumi desvio-padrão e documentei isso no código, mas vale conferir com o professor ou monitor se a intenção era outra. No Exercise 3, o teste ultrapassar levemente o intervalo [-1, 1] depois do escalonamento (item C/13) foi uma surpresa útil: mostra na prática por que "ajustar só no treino" é diferente de "garantir" um intervalo, mesmo fazendo tudo certo.

    Conclusão

    Os Exercises 1 e 2 mostram dois jeitos diferentes de uma rede neural "errar": no Exercise 1, o erro cresce continuamente à medida que as nuvens se espalham e se sobrepõem (um problema de quantidade de separação); no Exercise 2, o Dataset II mostra que a distância entre centros pode enganar completamente — classes com centros coincidentes podem ser perfeitamente separáveis se a fronteira certa não for um hiperplano (um problema de forma da fronteira). O Exercise 3 conecta isso a dados reais: a cauda pesada das colunas de gasto e a necessidade de evitar vazamento mostram que a maior parte do trabalho antes de treinar uma rede está em decidir como representar os dados, não no modelo em si.

    Exercise 1

    Abordagem

    Carreguei o MNIST via torchvision e dividi exatamente como especificado (50k treino / 10k validação via random_split com semente 42 / 10k teste oficial). O VAE (code/vae.py) segue a arquitetura dada — encoder 784→512→256 (ReLU), duas cabeças lineares 256→d para mu e log sigma^2, decoder d→256→512→784 (ReLU nas ocultas, sigmoide na saída) — com reparametrização, BCE de reconstrução e KL em forma fechada escritas à mão (nada de torch.distributions.kl_divergence ou classe de VAE pronta). Treinei com Adam (lr=1e-3, lotes de 128, 20 épocas), registrando reconstrução e KL separadamente no treino e na validação a cada época; o teste só entra nos números finais.

    """VAE — arquitetura e perda do enunciado, reutilizado sem alterações nos dois exercícios.

Encoder 784->512->256 (ReLU), duas cabeças lineares 256->d (mu, log-variância),
decoder d->256->512->784 (ReLU nas ocultas, sigmoide na saída). Reparametrização,
BCE de reconstrução e KL em forma fechada escritas à mão (sem
torch.distributions.kl_divergence e sem classe de VAE de biblioteca).
"""

import torch
import torch.nn as nn
import torch.nn.functional as F


class VAE(nn.Module):
    def __init__(self, d: int):
        super().__init__()
        self.d = d
        self.enc1 = nn.Linear(784, 512)
        self.enc2 = nn.Linear(512, 256)
        self.mu_head = nn.Linear(256, d)
        self.logvar_head = nn.Linear(256, d)
        self.dec1 = nn.Linear(d, 256)
        self.dec2 = nn.Linear(256, 512)
        self.dec3 = nn.Linear(512, 784)

    def encode(self, x: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor]:
        h = F.relu(self.enc1(x))
        h = F.relu(self.enc2(h))
        return self.mu_head(h), self.logvar_head(h)

    def reparameterize(self, mu: torch.Tensor, logvar: torch.Tensor) -> torch.Tensor:
        """z = mu + sigma * eps, eps ~ N(0, I) — o truque que deixa a amostragem diferenciável."""
        std = torch.exp(0.5 * logvar)
        eps = torch.randn_like(std)
        return mu + eps * std

    def decode(self, z: torch.Tensor) -> torch.Tensor:
        h = F.relu(self.dec1(z))
        h = F.relu(self.dec2(h))
        return torch.sigmoid(self.dec3(h))

    def forward(self, x: torch.Tensor):
        mu, logvar = self.encode(x)
        z = self.reparameterize(mu, logvar)
        x_hat = self.decode(z)
        return x_hat, mu, logvar


def vae_loss(x: torch.Tensor, x_hat: torch.Tensor, mu: torch.Tensor, logvar: torch.Tensor,
             beta: float = 1.0) -> tuple[torch.Tensor, torch.Tensor]:
    """Reconstrução (BCE, nats/imagem) e KL (forma fechada, nats/imagem), separados."""
    recon = F.binary_cross_entropy(x_hat, x, reduction="sum") / x.size(0)
    kl = 0.5 * torch.sum(mu.pow(2) + logvar.exp() - 1 - logvar) / x.size(0)
    return recon, kl

    """Exercise 1 — Um VAE no MNIST (d = 16).

Carrega e divide o MNIST (50k treino / 10k validação / 10k teste), treina o
VAE de ``vae.py`` por 20 épocas registrando reconstrução e KL separadamente,
e produz as Figuras 1-3.

Uso (a partir da raiz do repositório):

    python docs/exercises/vae/code/exercise1_mnist.py
"""

import sys
from pathlib import Path

import matplotlib.pyplot as plt
import torch
import torch.optim as optim
from torch.utils.data import DataLoader, random_split
from torchvision import datasets, transforms

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE))
from vae import VAE, vae_loss  # noqa: E402

FIGURES = HERE.parents[0] / "figures"
DATA_DIR = HERE / "data"

D_LATENT = 16
BATCH_SIZE = 128
EPOCHS = 20
LR = 1e-3


def load_data():
    transform = transforms.ToTensor()
    train_full = datasets.MNIST(str(DATA_DIR), train=True, download=True, transform=transform)
    test = datasets.MNIST(str(DATA_DIR), train=False, download=True, transform=transform)
    train, val = random_split(train_full, [50_000, 10_000], generator=torch.Generator().manual_seed(42))
    return train, val, test


def run_epoch(model: VAE, loader: DataLoader, optimizer=None) -> tuple[float, float]:
    """Uma passada pelo loader; treina se ``optimizer`` for dado, senão só avalia."""
    train_mode = optimizer is not None
    model.train(train_mode)
    total_recon, total_kl, n_batches = 0.0, 0.0, 0
    for x, _ in loader:
        x = x.view(x.size(0), -1)
        if train_mode:
            optimizer.zero_grad()
        x_hat, mu, logvar = model(x)
        recon, kl = vae_loss(x, x_hat, mu, logvar)
        if train_mode:
            (recon + kl).backward()
            optimizer.step()
        total_recon += recon.item()
        total_kl += kl.item()
        n_batches += 1
    return total_recon / n_batches, total_kl / n_batches


def plot_figure1(history: dict) -> None:
    fig, axes = plt.subplots(1, 2, figsize=(11, 4.5))
    epochs = range(1, EPOCHS + 1)
    axes[0].plot(epochs, history["train_recon"], label="Treino")
    axes[0].plot(epochs, history["val_recon"], label="Validação")
    axes[0].set_xlabel("Época")
    axes[0].set_ylabel("Reconstrução (nats/imagem)")
    axes[0].set_title("Termo de reconstrução")
    axes[0].legend()

    axes[1].plot(epochs, history["train_kl"], label="Treino")
    axes[1].plot(epochs, history["val_kl"], label="Validação")
    axes[1].set_xlabel("Época")
    axes[1].set_ylabel("KL (nats/imagem)")
    axes[1].set_title("Termo KL")
    axes[1].legend()

    fig.suptitle("Figura 1 — Reconstrução e KL por época (d = 16)")
    fig.tight_layout()
    fig.savefig(FIGURES / "fig01-loss-terms.png", dpi=150)
    plt.close(fig)


def plot_figure2(model: VAE, test) -> None:
    model.eval()
    digit_images = {}
    for x, label in test:
        if label not in digit_images:
            digit_images[label] = x.view(-1)
        if len(digit_images) == 10:
            break
    digit_images = [digit_images[d] for d in range(10)]
    X = torch.stack(digit_images)

    with torch.no_grad():
        mu, _ = model.encode(X)
        recon = model.decode(mu)

    fig, axes = plt.subplots(2, 10, figsize=(14, 3))
    for i in range(10):
        axes[0, i].imshow(X[i].view(28, 28), cmap="gray")
        axes[0, i].axis("off")
        axes[0, i].set_title(str(i), fontsize=9)
        axes[1, i].imshow(recon[i].view(28, 28), cmap="gray")
        axes[1, i].axis("off")
    axes[0, 0].set_ylabel("Original", fontsize=9)
    axes[1, 0].set_ylabel("Reconstrução", fontsize=9)
    fig.suptitle("Figura 2 — Original (cima) vs. reconstrução a partir de μ (baixo)")
    fig.tight_layout()
    fig.savefig(FIGURES / "fig02-reconstructions.png", dpi=150)
    plt.close(fig)


def plot_figure3(model: VAE) -> None:
    model.eval()
    torch.manual_seed(0)
    z = torch.randn(64, D_LATENT)
    with torch.no_grad():
        samples = model.decode(z)

    fig, axes = plt.subplots(8, 8, figsize=(8, 8))
    for i, ax in enumerate(axes.ravel()):
        ax.imshow(samples[i].view(28, 28), cmap="gray")
        ax.axis("off")
    fig.suptitle("Figura 3 — 64 amostras de z ~ N(0, I) (prior), decodificadas")
    fig.tight_layout()
    fig.savefig(FIGURES / "fig03-prior-samples.png", dpi=150)
    plt.close(fig)


def main() -> None:
    FIGURES.mkdir(parents=True, exist_ok=True)
    train, val, test = load_data()
    train_loader = DataLoader(train, batch_size=BATCH_SIZE, shuffle=True)
    val_loader = DataLoader(val, batch_size=BATCH_SIZE, shuffle=False)
    test_loader = DataLoader(test, batch_size=BATCH_SIZE, shuffle=False)

    torch.manual_seed(42)
    model = VAE(d=D_LATENT)
    optimizer = optim.Adam(model.parameters(), lr=LR)

    history = {"train_recon": [], "train_kl": [], "val_recon": [], "val_kl": []}
    for epoch in range(1, EPOCHS + 1):
        tr_recon, tr_kl = run_epoch(model, train_loader, optimizer)
        va_recon, va_kl = run_epoch(model, val_loader, optimizer=None)
        history["train_recon"].append(tr_recon)
        history["train_kl"].append(tr_kl)
        history["val_recon"].append(va_recon)
        history["val_kl"].append(va_kl)
        print(f"época {epoch:2d} | treino recon={tr_recon:7.3f} kl={tr_kl:6.3f} | "
              f"val recon={va_recon:7.3f} kl={va_kl:6.3f}")

    test_recon, test_kl = run_epoch(model, test_loader, optimizer=None)
    print(f"\nTeste (d={D_LATENT}): reconstrução={test_recon:.4f} nats, KL={test_kl:.4f} nats, "
          f"ELBO negativo={test_recon + test_kl:.4f} nats")
    print(f"KL de validação — primeira época: {history['val_kl'][0]:.4f}, "
          f"última época: {history['val_kl'][-1]:.4f}")

    plot_figure1(history)
    plot_figure2(model, test)
    plot_figure3(model)

    torch.save(model.state_dict(), HERE / "data" / "vae_d16.pt")


if __name__ == "__main__":
    main()

    Análise

    C — Treine

    Reconstrução no teste = 80,2365 nats, KL no teste = 20,5271 nats, ELBO negativo (soma) = 100,7636 nats — perto da referência do enunciado (~81 de reconstrução, ~20 de KL).

    E — Análise

    Figura 1: a KL sobe nas primeiras épocas enquanto a perda total cai. Por que o otimizador escolheria pagar mais KL?

    Porque o otimizador minimiza a soma dos dois termos, não cada um isoladamente. No início, com os pesos quase aleatórios, o posterior q(z|x) carrega pouca informação sobre x (está perto do prior, KL baixa) — e um código que não distingue bem as entradas produz reconstrução ruim (177 nats na época 1). Conforme o treino avança, "afastar" mu de zero e encolher sigma custa mais KL (sobe de 6,78 para 20,21 nats de treino — um aumento de ~13,4 nats), mas permite que o código carregue informação suficiente para reconstruir bem, derrubando a reconstrução de 177,28 para 80,09 nats — uma queda de ~97 nats. Como 97 >> 13,4, vale a pena pagar mais KL: a soma (o que de fato é minimizado) cai, mesmo que a KL isoladamente suba.

    A Figura 3 parece pior que a Figura 2 — dígitos mais borrados, alguns que não são dígito nenhum. Dê duas razões distintas.

    Onde as amostras do prior caem: N(0, I) espalha massa de forma isotrópica por todo o espaço latente de 16 dimensões, inclusive em regiões que o posterior aproximado dos dados reais raramente visita (o agregado dos q(z|x) de todas as imagens de treino não cobre o prior perfeitamente, sobretudo em 16 dimensões com treino limitado a 20 épocas) — uma amostra que cai numa região "entre" ou "fora" dos agrupamentos que os dígitos reais ocupam não corresponde a nenhum dígito bem definido.

    O que o decoder produz para um dado z: o decoder só foi treinado em z's vindos de posteriors de imagens reais; para um z de uma região pouco (ou nunca) vista no treino, a saída é extrapolação. Como a perda de reconstrução é uma verossimilhança de Bernoulli pixel a pixel (não há nenhum termo que force nitidez), o decoder tende a produzir uma média borrada de formas plausíveis em vez de comprometer-se com uma única forma nítida — o "borrão" característico do VAE.

    Por que decodificar mu e não um z amostrado na Figura 2?

    Porque a Figura 2 quer medir a qualidade da representação aprendida — o quanto o par encoder/decoder preserva da entrada —, não a variância extra que a amostragem z = mu + sigma*eps injetaria. mu é a estimativa pontual mais provável (a moda do posterior aproximado, que é gaussiano); decodificá-la dá a reconstrução determinística e menos ruidosa possível para aquela entrada. Usar um z amostrado misturaria "quão boa é a representação" com "quanta sorte teve essa amostra específica de ruído", dificultando a leitura da figura.

    """Exercise 2 — Um espaço latente que dá para ver (d = 2).

Mesmo modelo e orçamento de treino do Exercise 1, só mudando d=2. Produz as
Figuras 4-6 (dispersão do latente, plano decodificado via Phi^-1, e a
interpolação em pixels vs. em espaço latente usando o modelo d=16 salvo pelo
Exercise 1).

Uso (a partir da raiz do repositório, DEPOIS de rodar exercise1_mnist.py):

    python docs/exercises/vae/code/exercise2_latent2d.py
"""

import sys
from pathlib import Path

import matplotlib.pyplot as plt
import numpy as np
import torch
import torch.optim as optim
from scipy.stats import norm
from torch.utils.data import DataLoader
from torchvision import datasets, transforms

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE))
from vae import VAE, vae_loss  # noqa: E402
from exercise1_mnist import BATCH_SIZE, D_LATENT as D_EX1, DATA_DIR, EPOCHS, LR, load_data, run_epoch  # noqa: E402

FIGURES = HERE.parents[0] / "figures"
D_LATENT = 2


def train_2d_model():
    train, val, test = load_data()
    train_loader = DataLoader(train, batch_size=BATCH_SIZE, shuffle=True)
    val_loader = DataLoader(val, batch_size=BATCH_SIZE, shuffle=False)
    test_loader = DataLoader(test, batch_size=BATCH_SIZE, shuffle=False)

    torch.manual_seed(42)
    model = VAE(d=D_LATENT)
    optimizer = optim.Adam(model.parameters(), lr=LR)

    for epoch in range(1, EPOCHS + 1):
        tr_recon, tr_kl = run_epoch(model, train_loader, optimizer)
        va_recon, va_kl = run_epoch(model, val_loader, optimizer=None)
        print(f"época {epoch:2d} | treino recon={tr_recon:7.3f} kl={tr_kl:6.3f} | "
              f"val recon={va_recon:7.3f} kl={va_kl:6.3f}")

    test_recon, test_kl = run_epoch(model, test_loader, optimizer=None)
    print(f"\nTeste (d={D_LATENT}): reconstrução={test_recon:.4f} nats, KL={test_kl:.4f} nats")
    return model, test


def plot_figure4(model: VAE, test) -> np.ndarray:
    model.eval()
    loader = DataLoader(test, batch_size=512, shuffle=False)
    all_mu, all_y = [], []
    with torch.no_grad():
        for x, y in loader:
            mu, _ = model.encode(x.view(x.size(0), -1))
            all_mu.append(mu)
            all_y.append(y)
    mu = torch.cat(all_mu).numpy()
    y = torch.cat(all_y).numpy()

    fig, ax = plt.subplots(figsize=(7, 6))
    scatter = ax.scatter(mu[:, 0], mu[:, 1], c=y, cmap="tab10", s=4, alpha=0.6)
    ax.set_xlabel("$\\mu_1$")
    ax.set_ylabel("$\\mu_2$")
    ax.set_title("Figura 4 — Espaço latente 2D do conjunto de teste (10.000 imagens)")
    legend = ax.legend(*scatter.legend_elements(), title="Dígito", loc="center left",
                        bbox_to_anchor=(1.0, 0.5))
    fig.add_artist(legend)
    fig.tight_layout()
    fig.savefig(FIGURES / "fig04-latent-scatter.png", dpi=150)
    plt.close(fig)
    return mu


def plot_figure5(model: VAE) -> None:
    model.eval()
    q = np.linspace(0.05, 0.95, 15)
    grid_1d = norm.ppf(q)  # z = Phi^-1(q): mesma massa do prior em cada célula

    fig, axes = plt.subplots(15, 15, figsize=(9, 9))
    with torch.no_grad():
        for i, zi in enumerate(grid_1d):           # linha = eixo z2 (de cima para baixo)
            for j, zj in enumerate(grid_1d):        # coluna = eixo z1
                z = torch.tensor([[zj, zi]], dtype=torch.float32)
                img = model.decode(z).view(28, 28)
                axes[i, j].imshow(img, cmap="gray")
                axes[i, j].axis("off")
    fig.suptitle("Figura 5 — Plano latente decodificado (grade 15×15, espaçamento Φ⁻¹)")
    fig.tight_layout()
    fig.savefig(FIGURES / "fig05-latent-plane.png", dpi=150)
    plt.close(fig)


def plot_figure6(model_d16: VAE, test, idx_a: int, idx_b: int) -> None:
    model_d16.eval()
    x_a = test[idx_a][0].view(-1)
    x_b = test[idx_b][0].view(-1)

    with torch.no_grad():
        mu_a, _ = model_d16.encode(x_a.unsqueeze(0))
        mu_b, _ = model_d16.encode(x_b.unsqueeze(0))

    ts = np.linspace(0, 1, 10)
    fig, axes = plt.subplots(2, 10, figsize=(14, 3))
    with torch.no_grad():
        for i, t in enumerate(ts):
            pixel_interp = (1 - t) * x_a + t * x_b
            axes[0, i].imshow(pixel_interp.view(28, 28), cmap="gray")
            axes[0, i].axis("off")

            z_interp = (1 - t) * mu_a + t * mu_b
            latent_interp = model_d16.decode(z_interp).squeeze(0)
            axes[1, i].imshow(latent_interp.view(28, 28), cmap="gray")
            axes[1, i].axis("off")

    fig.suptitle(f"Figura 6 — Interpolação em pixels (cima) vs. em espaço latente d=16 (baixo), "
                 f"teste[{idx_a}] → teste[{idx_b}]")
    fig.tight_layout()
    fig.savefig(FIGURES / "fig06-interpolation.png", dpi=150)
    plt.close(fig)


def main() -> None:
    FIGURES.mkdir(parents=True, exist_ok=True)

    model, test = train_2d_model()
    plot_figure4(model, test)
    plot_figure5(model)

    Diga em uma frase por que esse espaçamento dá a cada célula a mesma massa do prior.

    Phi^-1(q) é a função quantil da normal padrão: por definição, intervalos igualmente espaçados em q (probabilidade acumulada) correspondem a intervalos de z que contêm a mesma massa de probabilidade sob N(0,1) — então espaçar q linearmente (em vez de z) faz cada célula da grade representar uma fatia igualmente provável do prior, em vez de superamostrar o centro ou desperdiçar células nas caudas raras.

    Quais dígitos se sobrepõem na Figura 4? Encontre a mesma transição na Figura 5 e descreva como são as imagens do meio do caminho.

    Na Figura 4, os dígitos 0, 1, 2 e 7 formam agrupamentos bem isolados nas bordas do plano; os dígitos 3, 4, 5, 6, 8 e 9 ficam todos amontoados numa região central comum, sem fronteiras claras entre si. Essa mesma mistura aparece na Figura 5: seguindo a diagonal central da grade (por volta das colunas 5-10), a imagem muda suavemente de um 2 para um 3, depois 5, 6, 4 e 9 — e nas células de transição (por exemplo entre as colunas que misturam 3/5/8) a imagem não é claramente nenhum dos dois: tem o laço de baixo de um 8 com o traço aberto de cima de um 5, ou a haste reta de um 4 já curvando no topo como um 9.

    d = 2 reconstrói claramente pior que d = 16. O que dois números por imagem obrigam o modelo a jogar fora? Use a Figura 5 como evidência.

    Dois números só têm "espaço" para codificar os eixos de variação mais grosseiros — principalmente qual família de formato o dígito tem (reto vs. com laço vs. arredondado, à julgar pela disposição espacial das famílias na Figura 5: 0/6 arredondados de um lado, 1/7 retos do outro). O que fica de fora é a identidade fina da classe e o estilo de traço: a Figura 5 mostra 3, 5, 6, 4, 8 e 9 ocupando uma faixa contínua e embaralhada no meio da grade — o modelo não tem dimensões sobrando para dar a cada um desses seis dígitos sua própria região bem separada, então eles compartilham vizinhança e se confundem. Com d = 16 (Figura 2) sobra espaço para manter os 10 dígitos separados e preservar detalhes de traço.

    Compare as duas linhas da Figura 6. Qual delas passa por imagens que parecem dígitos e por que a outra não passa?

    A linha de baixo (espaço latente, d = 16) passa por imagens que parecem dígitos o tempo todo — o 4 vai se curvando gradualmente até virar um 9 plausível, porque o decoder só precisa produzir saídas para pontos (1-t)*mu_a + t*mu_b que ficam relativamente perto da região do espaço latente que ele já aprendeu a decodificar em dígitos reais (a "variedade" aprendida). A linha de cima (pixels) produz, nos t intermediários, uma sobreposição fantasma dos dois dígitos — um 4 e um 9 visualmente somados, sem parecer nenhum dígito real — porque a média (1-t)*x_a + t*x_b é só uma mistura linear de intensidades de pixel, uma operação que não "sabe" nada sobre a forma dos dígitos e não respeita a estrutura não linear do que faz uma imagem parecer um traço contínuo e reconhecível.
